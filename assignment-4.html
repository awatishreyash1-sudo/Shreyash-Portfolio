const form = document.getElementById("marksheetForm");

form.addEventListener("submit", function(event) {

    event.preventDefault();

    const seatNumber =
        document.getElementById("seatNumber").value.trim();

    const message = document.getElementById("message");
    const marksheet = document.getElementById("marksheet");

    const student = students[seatNumber];

    marksheet.style.display = "none";

    if (!student) {

        message.textContent =
            "Invalid Seat Number! Student record not found.";

        message.style.display = "block";
        message.style.background = "#fff0f0";
        message.style.color = "#b42318";

        return;
    }

    message.textContent = "Marksheet found successfully!";
    message.style.display = "block";
    message.style.background = "#eaf8ef";
    message.style.color = "#207044";

    document.getElementById("studentDetails").innerHTML =
        "<p><strong>Seat Number:</strong> " +
        seatNumber + "</p>" +
        "<p><strong>Student Name:</strong> " +
        student.name + "</p>" +
        "<p><strong>Course:</strong> " +
        student.course + "</p>" +
        "<p><strong>Semester:</strong> " +
        student.semester + "</p>";

    let total = 0;
    let tableHTML = "";

    student.subjects.forEach(function(subject) {

        total += subject.marks;

        tableHTML +=
            "<tr><td>" + subject.name +
            "</td><td>100</td><td>" +
            subject.marks + "</td></tr>";

    });

    document.getElementById("marksTable").innerHTML = tableHTML;

    const percentage =
        (total / (student.subjects.length * 100)) * 100;

    const result = student.subjects.every(function(subject) {
        return subject.marks >= 35;
    }) ? "PASS" : "FAIL";

    document.getElementById("resultSummary").innerHTML =
        "<p><strong>Total Marks:</strong> " +
        total + " / " + (student.subjects.length * 100) +
        "</p>" +
        "<p><strong>Percentage:</strong> " +
        percentage.toFixed(2) + "%</p>" +
        "<p><strong>Result:</strong> " +
        result + "</p>";

    marksheet.style.display = "block";

});
Note: This is a client-side demonstration. Student records are stored in JavaScript and are not saved in a database.
