script.js

const factorialForm =
    document.getElementById("factorialForm");

const message =
    document.getElementById("message");

factorialForm.addEventListener("submit", function(event) {

    event.preventDefault();

    let number = document.getElementById("number").value;

    // Check empty input
    if (number.trim() === "") {
        message.textContent = "Please enter a number.";
        message.style.display = "block";
        return;
    }

    number = Number(number);

    // Check valid non-negative integer
    if (!Number.isInteger(number) || number < 0) {
        message.textContent =
            "Please enter a valid non-negative integer.";

        message.style.display = "block";
        return;
    }

    // Limit input to avoid numeric overflow
    if (number > 170) {
        message.textContent =
            "Please enter a number between 0 and 170.";

        message.style.display = "block";
        return;
    }

    // Calculate factorial
    let factorial = 1;

    for (let i = 1; i <= number; i++) {
        factorial = factorial * i;
    }

    // Display result
    message.textContent =
        "Factorial of " + number + " is " + factorial;

    message.style.display = "block";

});
Note: This program calculates factorial using a JavaScript for loop. The factorial of 0 is 1. The maximum supported input in this example is 170.
