script.js

const examForm = document.getElementById("examForm");
const message = document.getElementById("message");
const examDate = document.getElementById("examDate");

// Set today's date as minimum date
const today = new Date();

const localToday = new Date(
    today.getTime() - today.getTimezoneOffset() * 60000
).toISOString().split("T")[0];

examDate.min = localToday;

// Handle form submission
examForm.addEventListener("submit", function(event) {

    event.preventDefault();

    // Check form validation
    if (!examForm.checkValidity()) {
        examForm.reportValidity();
        return;
    }

    // Get student information
    const studentName =
        document.getElementById("studentName").value.trim();

    const rollNumber =
        document.getElementById("rollNumber").value.trim();

    const examName =
        document.getElementById("examName").value.trim();

    if (!studentName || !rollNumber || !examName) {
        message.textContent = "Please fill all required fields.";
        message.style.display = "block";
        return;
    }

    // Show success message
    message.textContent =
        "Registration Successful! Thank you, " +
        studentName +
        ". Your examination form has been validated successfully.";

    message.style.display = "block";

    // Reset form
    examForm.reset();
    examDate.min = localToday;

});
Note: This is a client-side demonstration. Form information is validated in the browser and is not saved to a database.
