script.js

function showDateTime() {

    // Create a Date object
    const currentDate = new Date();

    // Get current date
    const date = currentDate.toLocaleDateString();

    // Get current time
    const time = currentDate.toLocaleTimeString();

    // Display date and time in message box
    alert(
        "Current Date: " + date +
        "\nCurrent Time: " + time
    );

}
Note: This program uses JavaScript's built-in Date object to get the current date and time from the user's device.
