script.js

// Get HTML elements
const counter = document.getElementById("counter");
const status = document.getElementById("status");

const startBtn = document.getElementById("startBtn");
const stopBtn = document.getElementById("stopBtn");
const resetBtn = document.getElementById("resetBtn");

// Initialize variables
let count = 0;
let interval = null;

// Start Timer
startBtn.addEventListener("click", function() {

    // Prevent multiple intervals
    if (interval !== null) {
        return;
    }

    status.textContent = "Timer is running";

    // Execute code every 1000 milliseconds
    interval = setInterval(function() {

        count++;

        counter.textContent = count;

    }, 1000);

});

// Stop Timer
stopBtn.addEventListener("click", function() {

    // Stop the interval
    clearInterval(interval);

    interval = null;

    status.textContent = "Timer is stopped";

});

// Reset Timer
resetBtn.addEventListener("click", function() {

    // Stop running interval
    clearInterval(interval);

    interval = null;

    // Reset counter
    count = 0;

    counter.textContent = count;

    status.textContent = "Timer is reset";

});
Note: This is a client-side JavaScript demonstration. setInterval() repeatedly executes a function after a specified time interval, while clearInterval() stops the scheduled execution.
