Maximum of Two Numbers

Project Demonstration

Enter two numbers in the text boxes and click the Find Maximum button to display the greater number.

Maximum Number Finder

Enter two numbers below to find the maximum number.


How to test:
  • Enter the first number in the first text box.
  • Enter the second number in the second text box.
  • Click the Find Maximum button.
  • The greater number will be displayed below.

script.js

function findMaximum() {

    let num1 = Number(document.getElementById("firstNumber").value);

    let num2 = Number(document.getElementById("secondNumber").value);

    let result = document.getElementById("result");

    if (
        document.getElementById("firstNumber").value === "" ||
        document.getElementById("secondNumber").value === ""
    ) {
        alert("Please enter both numbers!");
        return;
    }

    if (num1 > num2) {
        result.textContent = "Maximum Number is: " + num1;
    }
    else if (num2 > num1) {
        result.textContent = "Maximum Number is: " + num2;
    }
    else {
        result.textContent = "Both numbers are equal: " + num1;
    }

}
Note: This program accepts two numbers using HTML input fields. JavaScript compares both numbers using if-else conditions and displays the maximum number.
