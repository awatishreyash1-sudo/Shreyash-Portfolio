script.js

function replaceCharacter() {

    let input = document.getElementById("userString").value;

    let result = document.getElementById("result");

    if (input === "") {
        alert("Please enter a string!");
        return;
    }

    let count = (input.match(/m/g) || []).length;

    let modifiedString = input.replace(/m/g, "a");

    result.innerHTML =
        "Original String: " + input +
        "<br>Modified String: " + modifiedString +
        "<br>Occurrences Found: " + count;
}
Note: This program uses JavaScript string methods. The match() method finds all occurrences of lowercase m, and replace() replaces every occurrence with a.
