script.js

// Get HTML elements

const clickButton =
    document.getElementById("clickButton");

const doubleButton =
    document.getElementById("doubleButton");

const mouseUpButton =
    document.getElementById("mouseUpButton");

const hoverBox =
    document.getElementById("hoverBox");

let eventCounter = 0;


// Update event activity log

function updateLog(eventName) {

    eventCounter++;

    document.getElementById("eventCount").textContent =
        "Total Events: " + eventCounter;

    document.getElementById("lastEvent").textContent =
        "Last Event: " + eventName;

}


// 1. onclick event

clickButton.addEventListener("click", function() {

    document.getElementById("clickStatus").textContent =
        "onclick event triggered successfully!";

    updateLog("onclick");

});


// 2. dblclick event

doubleButton.addEventListener("dblclick", function() {

    document.getElementById("doubleStatus").textContent =
        "dblclick event triggered successfully!";

    updateLog("dblclick");

});


// 3. mouseup event

mouseUpButton.addEventListener("mouseup", function() {

    document.getElementById("mouseUpStatus").textContent =
        "mouseup event triggered successfully!";

    updateLog("mouseup");

});


// 4. mouseover event

hoverBox.addEventListener("mouseover", function() {

    hoverBox.classList.add("active");

    hoverBox.textContent =
        "Mouse is over this box!";

    document.getElementById("hoverStatus").textContent =
        "mouseover event triggered successfully!";

    updateLog("mouseover");

});


// Mouse leaves hover box

hoverBox.addEventListener("mouseout", function() {

    hoverBox.classList.remove("active");

    hoverBox.textContent =
        "Move Mouse Pointer Here";

});


// Reset events

document.getElementById("resetEvents")
    .addEventListener("click", function() {

        eventCounter = 0;

        document.getElementById("eventCount").textContent =
            "Total Events: 0";

        document.getElementById("lastEvent").textContent =
            "Last Event: None";

        document.getElementById("clickStatus").textContent =
            "Waiting for click...";

        document.getElementById("doubleStatus").textContent =
            "Waiting for double click...";

        document.getElementById("mouseUpStatus").textContent =
            "Waiting for mouse release...";

        document.getElementById("hoverStatus").textContent =
            "Waiting for mouseover...";

        hoverBox.classList.remove("active");

        hoverBox.textContent =
            "Move Mouse Pointer Here";

    });
Note: This project demonstrates four JavaScript mouse events: onclick, dblclick, mouseup and mouseover. All events are handled directly in the browser using JavaScript.
